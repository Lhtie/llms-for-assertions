            path.push(new PathElem(cur, cp));
            i += Character.charCount(cp);
        }

        
        cur.isWord = false;
        size--;

            PathElem e = path.pop();
            e.node.pass--;

                if (e.incomingCp != null) {
                    parent.node.children.remove(e.incomingCp);
                }
            }
        }
        return true;
